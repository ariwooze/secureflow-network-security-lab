! SecureFlow Network Security Lab
! Device: Branch-Router
! Role: Branch gateway, IPsec VPN endpoint, AAA/RADIUS client, and Syslog client
! Environment: Cisco Packet Tracer
! Source: Network Security Audit Report by Ariana, Angela & Irdina
! Note: Local credentials, the RADIUS key, and the IPsec key are sanitised.
!
hostname Branch-Router
!
ip domain-name secureflow.local
username admin privilege 15 secret <REDACTED_ADMIN_SECRET>
username backup privilege 15 secret <REDACTED_BACKUP_SECRET>
ip ssh version 2
ip ssh authentication-retries 5
ip ssh time-out 60
!
aaa new-model
aaa authentication login default group radius local
aaa authorization exec default group radius local
aaa accounting exec default start-stop group radius
!

radius-server host 192.168.1.10 key <REDACTED_RADIUS_KEY>
